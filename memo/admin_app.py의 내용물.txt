import os
import streamlit as st
import streamlit_authenticator as stauth
import sheet_manager as sm

# 로컬 환경용 dotenv 로드 (클라우드 환경에서 python-dotenv가 없어도 ModuleNotFoundError 방지)
try:
    from dotenv import load_dotenv
    load_dotenv()
except ImportError:
    pass

st.set_page_config(
    page_title="생활관 자치회 물품 대여 관리자 포털",
    page_icon="🔐",
    layout="wide"
)

# 1. 환경변수 / Secrets 연동 관리자 인증 설정 (안전한 get_env_val)
def get_env_val(key, default=""):
    val = default
    try:
        # Streamlit Cloud 배포 환경 또는 secrets.toml 확인
        if key in st.secrets:
            val = st.secrets[key]
        else:
            val = os.getenv(key, default)
    except Exception:
        # secrets.toml 부재로 인한 StreamlitSecretNotFoundError 발생 시 .env 조회
        val = os.getenv(key, default)
        
    # 따옴표 기호가 포함되어 있을 경우 안전하게 제거
    if isinstance(val, str):
        val = val.strip().strip("\"'")
    return val

# .env 또는 Secrets에서 관리자 계정 정보 불러오기
admin1_id = get_env_val("ADMIN1_ID", "admin1")
admin1_name = get_env_val("ADMIN1_NAME", "생활관 자치회원1")
admin1_pw = get_env_val("ADMIN1_PW", "dorm1234")
admin1_email = get_env_val("ADMIN1_EMAIL", "scnu.dormitory@gmail.com")

admin2_id = get_env_val("ADMIN2_ID", "admin2")
admin2_name = get_env_val("ADMIN2_NAME", "생활관 자치회원2")
admin2_pw = get_env_val("ADMIN2_PW", "admin1234")
admin2_email = get_env_val("ADMIN2_EMAIL", "master@dormitory.com")

cookie_key = get_env_val("COOKIE_KEY", "dorm_super_secret_cookie_key_2026")

# 비밀번호 해싱
try:
    hasher = stauth.Hasher()
    hashed_pw1 = hasher.hash(admin1_pw)
    hashed_pw2 = hasher.hash(admin2_pw)
except Exception:
    hashed_pw1 = stauth.Hasher([admin1_pw]).generate()[0]
    hashed_pw2 = stauth.Hasher([admin2_pw]).generate()[0]

credentials = {
    "usernames": {
        admin1_id: {"name": admin1_name, "password": hashed_pw1, "email": admin1_email},
        admin2_id: {"name": admin2_name, "password": hashed_pw2, "email": admin2_email}
    }
}

authenticator = stauth.Authenticate(
    credentials,
    cookie_name="dorm_admin_auth_cookie",
    key=cookie_key,
    cookie_expiry_days=1
)

try:
    authenticator.login(location="main")
except TypeError:
    authenticator.login("main")

if st.session_state.get("authentication_status") is False:
    st.error("❌ 아이디 또는 비밀번호가 올바르지 않습니다.")
    st.stop()
elif st.session_state.get("authentication_status") is None:
    st.warning("🔒 생활관 자치회 관리자 전용 포털입니다. 로그인 정보를 입력해주세요.")
    st.stop()

# 2. 관리자 메인 화면
with st.sidebar:
    st.write(f"👤 접속 관리자: **{st.session_state.get('name')}**님")
    try:
        authenticator.logout(button_name="로그아웃", location="sidebar")
    except TypeError:
        authenticator.logout("로그아웃", "sidebar")

st.title("🏢 생활관 자치회 물품 대여 관리 센터")
st.markdown("---")

tab1, tab2, tab3 = st.tabs(["🔔 승인 / 반려 대기", "📦 물품 전달 및 반납 관리", "📊 전체 내역"])

try:
    all_rentals = sm.get_all_rentals()
except Exception:
    all_rentals = []

# [탭 1: 승인 / 반려 처리]
with tab1:
    st.subheader("신규 대여 신청 검토")
    pending = [r for r in all_rentals if r.get("status") == "승인대기"]
    
    if not pending:
        st.info("현재 대기 중인 신청 건이 없습니다.")
    else:
        for idx, r in enumerate(pending):
            rental_id = r.get("rental_id", "")
            item_name = r.get("item_name", "물품")
            student_name = r.get("student_name", "미상")
            student_id = r.get("student_id", "-")
            building_name = r.get("building_name", "")
            room_no = r.get("room_no", "-")
            req_time = r.get("req_time", "-")
            email = r.get("email", "미입력")
            agreement = r.get("agreement", "-")
            expected_return = r.get("expected_return", "-")
            
            location_info = f"{building_name} {room_no}호" if building_name else f"{room_no}호"
            desired_date = r.get("desired_date", "").strip()
            display_desired = desired_date if desired_date else "-"

            with st.container(border=True):
                col_info, col_action = st.columns([1.2, 1.8])
                with col_info:
                    st.markdown(f"**#{idx+1} {item_name}**")
                    st.write(f"- 신청자: **{student_name}** ({student_id})")
                    st.write(f"- 소속/연락처: {location_info} / {email}")
                    st.write(f"- 신청접수시각: {req_time}")
                    st.write(f"- 관생 희망일시: `{display_desired}`")
                    st.write(f"- 반납 예정일자: `{expected_return}`")
                    st.write(f"- 서약 동의: **{agreement}**")
                
                with col_action:
                    edit_desired = st.text_input(
                        "대여 희망일시 (관리자 수정 가능)",
                        value=display_desired,
                        key=f"edit_desired_{rental_id}"
                    )
                    c_date = st.text_input(
                        "확정 대여일시 안내 문구", 
                        value=f"{edit_desired} 생활관 자치회실 방문 수령", 
                        key=f"c_{rental_id}"
                    )
                    c_memo = st.text_input(
                        "관생 안내 메모 / 반려 사유", 
                        placeholder="예: 시간 준수 요망", 
                        key=f"m_{rental_id}"
                    )
                    
                    btn_col1, btn_col2, _ = st.columns([1, 1, 1.5])
                    with btn_col1:
                        if st.button("🟢 승인", key=f"app_{rental_id}", use_container_width=True):
                            with st.spinner("승인 처리 중..."):
                                success = sm.approve_rental(
                                    rental_id=rental_id, item_name=item_name,
                                    confirmed_date=c_date, admin_memo=c_memo,
                                    updated_desired_date=edit_desired
                                )
                                # 📩 사생에게 승인 안내 메일 발송
                                if success and email and email != "미입력":
                                    appr_subj = f"[생활관 자치회] {item_name} 대여 신청이 승인되었습니다."
                                    appr_body = f"""
                                    <div style="font-family: Arial, sans-serif; line-height: 1.6; color: #333;">
                                        <h2 style="color: #2e7d32;">✅ 물품 대여 신청 승인 완료</h2>
                                        <p><b>{student_name}</b>님, 신청하신 물품 대여가 최종 승인되었습니다.</p>
                                        <div style="background-color: #f1f8e9; border-left: 5px solid #4caf50; padding: 12px 16px; margin: 15px 0;">
                                            <p style="margin: 0; font-weight: bold; color: #1b5e20;">📍 수령 안내: {c_date}</p>
                                            {f'<p style="margin: 5px 0 0 0; color: #333;">💬 관리자 메모: {c_memo}</p>' if c_memo else ''}
                                        </div>
                                        <p>안내된 수령 일시에 맞춰 생활관 자치회실로 방문해 주시기 바랍니다.</p>
                                    </div>
                                    """

                            if success:

                                    rej_subj = f"[생활관 자치회] {item_name} 대여 신청이 반려되었습니다."
                                    rej_body = f"""
                                    <div style="font-family: Arial, sans-serif; line-height: 1.6; color: #333;">
                                        <h2 style="color: #c62828;">❌ 물품 대여 신청 반려 안내</h2>

                            st.rerun()

# [탭 2: 수령 및 반납 처리]

    approved = [r for r in all_rentals if r.get("status") == "승인완료"]

    else:
        for r in approved:
            rental_id = r.get("rental_id", "")
            item_name = r.get("item_name", "물품")
            student_name = r.get("student_name", "미상")
            confirmed_date = r.get("confirmed_date", "-")

            with st.container(border=True):
                c1, c2, c3 = st.columns([2, 2, 1])
                c1.write(f"**{item_name}** | {student_name}")
                c2.write(f"수령 약속: {confirmed_date}")
                if c3.button("물품 인출 완료", key=f"take_{rental_id}", use_container_width=True):
                    sm.start_rental(rental_id)
                    st.rerun()

    st.markdown("---")
    st.subheader("2. 물품 반납 처리 (대여중 $\\rightarrow$ 반납완료)")
    renting = [r for r in all_rentals if r.get("status") == "대여중"]

    else:
        for r in renting:

            item_name = r.get("item_name", "물품")
            student_name = r.get("student_name", "미상")
            expected_return = r.get("expected_return", "-")
            confirmed_date = r.get("confirmed_date", "-")

            with st.container(border=True):
                c1, c2, c3 = st.columns([2, 2, 1])
                c1.write(f"**{item_name}** | {student_name}")
                c2.write(f"대여일: {confirmed_date} / 반납 예정일: {expected_return}")
                if c3.button("반납 완료", key=f"ret_{rental_id}", use_container_width=True):
